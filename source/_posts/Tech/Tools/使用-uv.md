环境配置：

```shell
setx UV_CACHE_DIR "D:\xxx\uv\cache" /M
setx UV_CREDENTIALS_DIR "D:\xxx\uv\credentials" /M
setx UV_INSTALL_DIR "D:\xxx\uv\install" /M
setx UV_PYTHON_BIN_DIR "D:\xxx\uv\python_bin" /M
setx UV_PYTHON_CACHE_DIR "D:\xxx\uv\python_cache" /M
setx UV_PYTHON_INSTALL_DIR "D:\xxx\uv\python_install" /M
setx UV_TOOL_BIN_DIR "D:\xxx\uv\tool_bin" /M
setx UV_TOOL_DIR "D:\xxx\uv\tool" /M
setx UV_WORKING_DIR "D:\xxx\uv\working" /M
setx UV_INDEX_URL "https://pypi.tuna.tsinghua.edu.cn/simple" /M
setx UV_PYTHON_INSTALL_MIRROR "https://registry.npmmirror.com/-/binary/python-build-standalone/" /M
```
